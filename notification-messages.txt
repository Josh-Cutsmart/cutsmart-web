# CutSmart notification wording
#
# Change the words after "title:", "message:" and so on. Keep the word before the colon as it is.
# Words in {curly brackets} fill in automatically. You can move them or leave them out, but only use ones
# already listed under that notification.
# If something is missing or a bracket word isn't recognised, CutSmart uses its original wording for that line.
# Lines starting with # are notes and are ignored.
# Changes go live when you push. Notifications already sent keep the wording they were sent with.
#
#
# ---------- PLACEHOLDERS ----------
#
# What each {word} turns into, with an example. The sections below say which ones each notification can use.
#
# People
#   {person}       A teammate's name                               Mere Tawhiri
#   {client}       The client's name                               Aroha Williams
#   {staff}        The staff member who did it for the client      Alex Morgan
#   {sender}       The email address a lead came from              jane@example.com
#   {name}         A lead's name, or the project or lead name
#                  inside the {thing} wording                      Jane Smith
#
# Projects and leads
#   {project}      The project's name                              Harbour View Kitchen
#   {lead}         The lead's name                                 Jane Smith
#   {thing}        The project or lead, worded by the
#                  [column_automation.thing] section               "Harbour View Kitchen"
#   {column}       A board column                                  Production
#   {new_column}   The column it moved to                          Site Measure
#   {time}         How long it's been in a column                  3 days
#   {role}         Their access on a project                       an editor / a viewer
#   {hours}        How long production is unlocked for             2 hours
#
# Client portal
#   {price}        The quote's total                               $18,450.00
#   {reason}       Why the client declined                         Going with a different layout
#   {accepted}     How many specifications were accepted           12
#   {total}        How many specifications there were              15
#   {ago}          How long ago it was sent                        2 days ago
#
# Calendar
#   {event}        The event's title                               Benchtop install
#   {when}         How soon it is                                  today / tomorrow / in 2 days
#
# Your company
#   {version}      The CutSmart version                            v0.6.1
#   {company}      The company's name                              Taylor Joinery
#   {old_role}     Their role before                               Staff
#   {new_role}     Their role now                                  Manager
#   {code}         The access code                                 AB12-CD34
#   {label}        The name given to that access code              New starter
#
# Several together
#   {count}        How many                                        3


# ---------- PROJECTS ----------

# Someone is given access to a project for the first time. {role} is "an editor" or "a viewer".
[project_access.added]
title: Project
message: {person} added you to "{project}" as {role}.

# Someone's access changes between viewer and editor.
[project_access.changed]
title: Project
message: {person} made you {role} on "{project}".

[project_assigned]
title: Project
message: You were assigned to "{project}".

# A column automation assigned it.
[project_assigned.automation]
title: Project
message: "{project}" was assigned to you.

[project_unassigned]
title: Project
message: You were removed from "{project}".

# The buttons only show on Android and computers.
[production_unlock_request]
title: Project
message: {person} asked to unlock production on "{project}".
button_accept: Accept
button_decline: Decline

# {hours} reads like "2 hours".
[production_unlock_response.accepted]
title: Project
message: {person} accepted your request — you can edit production on "{project}" for {hours}.

[production_unlock_response.declined]
title: Project
message: {person} declined your request to unlock production on "{project}".

# What shows after tapping Accept or Decline on the notification itself. {person} asked, {hours} reads like "2 hours".
[production_unlock_tap.accepted]
title: Project
message: Accepted — {person} can edit production for {hours}.

[production_unlock_tap.declined]
title: Project
message: Declined — {person} has been told.

[production_unlock_tap.already_answered]
title: Project
message: Someone has already answered this request.


# ---------- LEADS ----------

[lead_new.email]
title: New Lead
message: From Email: {sender}

[lead_new.form]
title: New Lead
message: From Form: {name}

[lead_new.manual]
title: New Lead
message: From Create Lead: {name}

# Zapier, or anything else.
[lead_new.other]
title: New Lead
message: From Zapier: {name}

[lead_access]
title: Leads
message: {person} added you to the lead "{lead}".

# A column automation assigned it.
[lead_access.automation]
title: Leads
message: The lead "{lead}" was assigned to you.


# ---------- COLUMN AUTOMATIONS ----------

# {thing} is one of the two wordings below, depending on whether it's a project or a lead.
[column_automation.thing]
project: "{name}"
lead: The lead "{name}"

[column_automation.enters]
title: Moved into {column}
message: {thing} moved into {column}.

[column_automation.leaves]
title: Moved out of {column}
message: {thing} moved out of {column} — now in {new_column}.

# Leaves, when where it went isn't known.
[column_automation.leaves_unknown]
title: Moved out of {column}
message: {thing} moved out of {column}.

# {time} reads like "3 days".
[column_automation.stays]
title: Still in {column}
message: {thing} has been in {column} for {time}.

# Told to whoever's move set off a move automation.
[column_automation.moved_on]
title: Moved into {column}
message: {thing} moved into {column}.

# The same, when a lead moved on to the Dashboard.
[column_automation.became_project]
title: Moved into {column}
message: {thing} moved into {column} and became a project.


# ---------- CLIENT PORTAL ----------

[quote_accepted]
title: Quote accepted for {project}
message: {client} accepted your quote for {price}

# The quote has no total marked.
[quote_accepted.no_price]
title: Quote accepted for {project}
message: {client} accepted your quote

[quote_accepted.on_behalf]
title: Quote accepted for {project}
message: Accepted on behalf of {client} by {staff} — {price}

[quote_accepted.on_behalf_no_price]
title: Quote accepted for {project}
message: Accepted on behalf of {client} by {staff}

[quote_declined]
title: Quote declined
message: {client} declined quote for {project}: "{reason}"

[quote_declined.no_reason]
title: Quote declined
message: {client} declined quote for {project}

[specs_submitted]
title: Specifications Submitted for {project}
message: {client} submitted the specifications. ({accepted}/{total}) accepted

# The sheet has nothing to confirm.
[specs_submitted.no_confirmations]
title: Specifications Submitted for {project}
message: {client} submitted the specifications

[specs_submitted.on_behalf]
title: Specifications Submitted for {project}
message: Submitted on behalf of {client} by {staff}

[quote_sent]
title: Quote sent to client
message: {person} sent a Quote for "{project}" to the client.

[specs_sent]
title: Specifications sent to client
message: {person} sent Specifications for "{project}" to the client.

# {ago} reads like "2 days ago".
[quote_pending]
title: Quote not accepted yet
message: {client} hasn't accepted the quote for {project} yet — sent {ago}.

[specs_pending]
title: Specifications not submitted yet
message: {client} hasn't submitted the specifications for {project} yet — sent {ago}.


# ---------- CALENDAR ----------

# {when} reads like "today", "tomorrow" or "in 2 days".
[calendar_reminder]
title: Upcoming: {event}
message: "{event}" for {project} is {when}.

[calendar_reminder.no_project]
title: Upcoming: {event}
message: "{event}" is {when}.


# ---------- CUTSMART ----------

# phone: what the phone shows. message: what's kept when a version has no notes.
[app_version]
title: New version {version}
message: CutSmart has been updated.
phone: See what's new in CutSmart.

[role_change]
title: Role updated
message: Your role in {company} changed from "{old_role}" to "{new_role}".

[access_code_used]
title: Access code used
message: {person} joined with your access code {code}.

# The code has a name.
[access_code_used.with_label]
title: Access code used
message: {person} joined with your access code {code} ({label}).


# ---------- SEVERAL TOGETHER ----------

# When several are sent as one. Used by "Time between notifications", events due at the same moment, and ones that reach a
# phone late.
[group]
mixed_title: {count} new notifications
more: and {count} more

# How each kind reads in a group's title, one | several. They show after the count, like "3 new leads".

project_access: project shared with you | projects shared with you

lead_new: new lead | new leads
lead_access: lead assigned to you | leads assigned to you
quote_accepted: quote accepted | quotes accepted
quote_declined: quote declined | quotes declined
specs_submitted: specification submitted | specifications submitted
quote_sent: quote sent | quotes sent
specs_sent: specification sent | specifications sent
calendar_reminder: new event coming up | new events coming up
quote_pending: quote waiting on a client | quotes waiting on clients
specs_pending: specification waiting on a client | specifications waiting on clients

role_change: role change | role changes
